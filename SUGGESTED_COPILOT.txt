<?php
// App settings loaded from DB; falls back to sane defaults.
function app_setting(PDO $db, string $key, $default=null) {
    $db->exec("CREATE TABLE IF NOT EXISTS app_settings (skey VARCHAR(100) PRIMARY KEY, sval TEXT)");
    $st=$db->prepare("SELECT sval FROM app_settings WHERE skey=?"); $st->execute([$key]);
    $row=$st->fetch(); return $row? $row['sval'] : $default;
}
function set_app_setting(PDO $db, string $key, string $val) {
    $db->exec("CREATE TABLE IF NOT EXISTS app_settings (skey VARCHAR(100) PRIMARY KEY, sval TEXT)");
    $st=$db->prepare("INSERT INTO app_settings(skey,sval) VALUES(?,?) ON DUPLICATE KEY UPDATE sval=VALUES(sval)");
    $st->execute([$key,$val]);
}

<?php
require_once __DIR__ . '/settings.php';

// Decide CGST/SGST vs IGST based on org_state vs place_of_supply
function gst_split(string $org_state, string $pos, float $taxable, float $rate): array {
    $gst = round($taxable * $rate / 100, 2);
    if ($org_state === $pos) {
        $cgst = round($gst/2, 2);
        $sgst = $gst - $cgst;
        return ['cgst'=>$cgst,'sgst'=>$sgst,'igst'=>0.0,'total'=>$taxable+$gst];
    }
    return ['cgst'=>0.0,'sgst'=>0.0,'igst'=>$gst,'total'=>$taxable+$gst];
}

<?php
require_once __DIR__ . '/db_connect.php';
require_once __DIR__ . '/accounting_lib.php';
require_once __DIR__ . '/tax_rules.php';
require_once __DIR__ . '/settings.php';

function org_state(PDO $db){ return app_setting($db,'org_state','UP'); }
function bank_ac(){ return '1010'; } // default bank account code
function ar_ac(){ return '1200'; }
function ap_ac(){ return '2100'; }
function sales_ac(){ return '4000'; }
function purchases_ac(){ return '5000'; }
function input_cgst(){ return '1402'; }
function input_sgst(){ return '1403'; }
function input_igst(){ return '1404'; }
function output_cgst(){ return '2102'; }
function output_sgst(){ return '2103'; }
function output_igst(){ return '2104'; }

function ensure_extended_coa(PDO $db){
    acc_init_coa($db);
    // Add specific GST accounts if missing
    $adds = [
        ['1402','Input CGST','1000'],
        ['1403','Input SGST','1000'],
        ['1404','Input IGST','1000'],
        ['2102','Output CGST','2000'],
        ['2103','Output SGST','2000'],
        ['2104','Output IGST','2000'],
    ];
    $ins=$db->prepare("INSERT IGNORE INTO accounts(code,name,parent_code) VALUES(?,?,?)");
    foreach($adds as $a){ $ins->execute($a); }
    acc_migrate_ledgers($db);
}

// Sales invoice posting: Dr AR or Bank, Cr Sales, Cr Output GST*
function post_sales(PDO $db, int $invoice_id){
    ensure_extended_coa($db);
    $iv = $db->prepare("SELECT * FROM sales_invoices WHERE id=? AND posted_at IS NULL");
    $iv->execute([$invoice_id]); $i=$iv->fetch(); if(!$i) return false;

    $org = org_state($db);
    $pos = $i['place_of_supply'];
    $taxable = (float)$i['taxable'];
    $cgst=(float)$i['cgst']; $sgst=(float)$i['sgst']; $igst=(float)$i['igst'];

    $lines = [
        ['account_code'=>ar_ac(), 'debit'=>$i['total'], 'credit'=>0],
        ['account_code'=>sales_ac(), 'debit'=>0, 'credit'=>$taxable],
    ];
    if ($cgst>0) $lines[] = ['account_code'=>output_cgst(),'debit'=>0,'credit'=>$cgst];
    if ($sgst>0) $lines[] = ['account_code'=>output_sgst(),'debit'=>0,'credit'=>$sgst];
    if ($igst>0) $lines[] = ['account_code'=>output_igst(),'debit'=>0,'credit'=>$igst];

    $eid = acc_post_journal($db, $i['idate'], "Sales Inv #{$i['inv_no']}", $lines);
    $st=$db->prepare("UPDATE sales_invoices SET posted_at=NOW(), je_id=? WHERE id=?"); $st->execute([$eid,$invoice_id]);
    return $eid;
}

// Purchase posting: Dr Purchases, Dr Input GST*, Cr AP
function post_purchase(PDO $db, int $invoice_id){
    ensure_extended_coa($db);
    $iv = $db->prepare("SELECT * FROM purchase_invoices WHERE id=? AND posted_at IS NULL");
    $iv->execute([$invoice_id]); $i=$iv->fetch(); if(!$i) return false;

    $lines = [
        ['account_code'=>purchases_ac(), 'debit'=>$i['taxable'], 'credit'=>0],
    ];
    if ($i['cgst']>0) $lines[] = ['account_code'=>input_cgst(),'debit'=>$i['cgst'],'credit'=>0];
    if ($i['sgst']>0) $lines[] = ['account_code'=>input_sgst(),'debit'=>$i['sgst'],'credit'=>0];
    if ($i['igst']>0) $lines[] = ['account_code'=>input_igst(),'debit'=>$i['igst'],'credit'=>0];

    $lines[] = ['account_code'=>ap_ac(), 'debit'=>0,'credit'=>$i['total']];

    $eid = acc_post_journal($db, $i['idate'], "Purchase Bill #{$i['bill_no']}", $lines);
    $st=$db->prepare("UPDATE purchase_invoices SET posted_at=NOW(), je_id=? WHERE id=?"); $st->execute([$eid,$invoice_id]);
    return $eid;
}

// Receipt: Dr Bank, Cr AR
function post_receipt(PDO $db, int $rid){
    ensure_extended_coa($db);
    $r = $db->prepare("SELECT * FROM receipts WHERE id=? AND posted_at IS NULL");
    $r->execute([$rid]); $x=$r->fetch(); if(!$x) return false;
    $eid = acc_post_journal($db, $x['rdate'], "Receipt #$rid", [
        ['account_code'=>bank_ac(), 'debit'=>$x['amount'],'credit'=>0],
        ['account_code'=>ar_ac(),   'debit'=>0,'credit'=>$x['amount']],
    ]);
    $st=$db->prepare("UPDATE receipts SET posted_at=NOW(), je_id=? WHERE id=?"); $st->execute([$eid,$rid]);
    return $eid;
}

// Payment: Dr AP, Cr Bank
function post_payment(PDO $db, int $pid){
    ensure_extended_coa($db);
    $p = $db->prepare("SELECT * FROM payments WHERE id=? AND posted_at IS NULL");
    $p->execute([$pid]); $x=$p->fetch(); if(!$x) return false;
    $eid = acc_post_journal($db, $x['pdate'], "Payment #$pid", [
        ['account_code'=>ap_ac(),   'debit'=>$x['amount'],'credit'=>0],
        ['account_code'=>bank_ac(), 'debit'=>0,'credit'=>$x['amount']],
    ]);
    $st=$db->prepare("UPDATE payments SET posted_at=NOW(), je_id=? WHERE id=?"); $st->execute([$eid,$pid]);
    return $eid;
}

// Credit note: Dr Sales Returns (use 5001), Dr Output GST*, Cr AR
function post_credit_note(PDO $db, int $cid){
    ensure_extended_coa($db);
    // Ensure Sales Returns account
    $db->prepare("INSERT IGNORE INTO accounts(code,name,parent_code) VALUES(?,?,?)")->execute(['5001','Sales Returns','5000']);
    $c = $db->prepare("SELECT * FROM credit_notes WHERE id=? AND posted_at IS NULL");
    $c->execute([$cid]); $x=$c->fetch(); if(!$x) return false;

    $lines = [
        ['account_code'=>'5001','debit'=>$x['taxable'],'credit'=>0],
    ];
    if ($x['cgst']>0) $lines[] = ['account_code'=>output_cgst(),'debit'=>$x['cgst'],'credit'=>0];
    if ($x['sgst']>0) $lines[] = ['account_code'=>output_sgst(),'debit'=>$x['sgst'],'credit'=>0];
    if ($x['igst']>0) $lines[] = ['account_code'=>output_igst(),'debit'=>$x['igst'],'credit'=>0];
    $lines[] = ['account_code'=>ar_ac(),'debit'=>0,'credit'=>$x['total']];

    $eid = acc_post_journal($db, $x['cdate'], "Credit Note #$cid", $lines);
    $st=$db->prepare("UPDATE credit_notes SET posted_at=NOW(), je_id=? WHERE id=?"); $st->execute([$eid,$cid]);
    return $eid;
}

// Debit note: Dr AP, Cr Purchase Returns (use 4001) + Input GST reversal
function post_debit_note(PDO $db, int $did){
    ensure_extended_coa($db);
    $db->prepare("INSERT IGNORE INTO accounts(code,name,parent_code) VALUES(?,?,?)")->execute(['4001','Purchase Returns','4000']);
    $d = $db->prepare("SELECT * FROM debit_notes WHERE id=? AND posted_at IS NULL");
    $d->execute([$did]); $x=$d->fetch(); if(!$x) return false;

    $lines = [
        ['account_code'=>ap_ac(),'debit'=>0,'credit'=>$x['total']],
        ['account_code'=>'4001','debit'=>0,'credit'=>$x['taxable']],
    ];
    if ($x['cgst']>0) $lines[] = ['account_code'=>input_cgst(),'debit'=>0,'credit'=>$x['cgst']];
    if ($x['sgst']>0) $lines[] = ['account_code'=>input_sgst(),'debit'=>0,'credit'=>$x['sgst']];
    if ($x['igst']>0) $lines[] = ['account_code'=>input_igst(),'debit'=>0,'credit'=>$x['igst']];

    $st=$db->prepare("UPDATE debit_notes SET posted_at=NOW(), je_id=? WHERE id=?"); $st->execute([$eid,$did]);
