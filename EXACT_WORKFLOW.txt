### app/Core/DB.php

```php
<?php
namespace App\Core;

use PDO;

class DB {
  private static ?PDO $pdo = null;

  public static function pdo(): PDO {
    if (!self::$pdo) {
      $cfg = require __DIR__ . '/../../config/database.php';
      self::$pdo = new PDO($cfg['dsn'], $cfg['user'], $cfg['pass'], $cfg['options']);
      self::$pdo->exec("SET time_zone = '+05:30'");
    }
    return self::$pdo;
  }

  public static function tx(callable $fn) {
    $pdo = self::pdo();
    try {
      $pdo->beginTransaction();
      $result = $fn($pdo);
      $pdo->commit();
      return $result;
    } catch (\Throwable $e) {
      if ($pdo->inTransaction()) $pdo->rollBack();
      throw $e;
    }
  }
}
```

### app/Core/Router.php (minimal, fast)

```php
<?php
namespace App\Core;

class Router {
  private array $routes = ['GET'=>[], 'POST'=>[]];

  public function get(string $path, $handler) { $this->routes['GET'][$path] = $handler; }
  public function post(string $path, $handler) { $this->routes['POST'][$path] = $handler; }

  public function dispatch() {
    $method = $_SERVER['REQUEST_METHOD'];
    $uri = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH);
    foreach ($this->routes[$method] as $pattern => $handler) {
      $regex = '#^' . preg_replace('#\{[^/]+\}#', '([^/]+)', $pattern) . '$#';
      if (preg_match($regex, $uri, $m)) {
        array_shift($m);
        return $this->call($handler, $m);
      }
    }
    http_response_code(404);
    echo "404 Not Found";
  }

  private function call($handler, array $params) {
    if (is_array($handler)) {
      [$class, $method] = $handler;
      $obj = new $class;
      return $obj->$method(...$params);
    }
    return $handler(...$params);
  }
}
```

## Core GST domain logic

### Tax split helpers

```php
<?php
namespace App\Helpers;

class Money {
  public static function round2(float $v): float { return round($v, 2, PHP_ROUND_HALF_UP); }
}

class GST {
  // Determine intra vs inter state. You can adapt based on your party/place_of_supply columns.
  public static function split(float $taxAmount, bool $intra): array {
    if ($intra) {
      $cgst = Money::round2($taxAmount / 2);
      $sgst = Money::round2($taxAmount - $cgst);
      return ['cgst'=>$cgst, 'sgst'=>$sgst, 'igst'=>0.0];
    } else {
      return ['cgst'=>0.0, 'sgst'=>0.0, 'igst'=>Money::round2($taxAmount)];
    }
  }

  public static function computeLine(float $qty, float $rate, float $gstRatePct): array {
    $amount = Money::round2($qty * $rate);
    $tax = Money::round2($amount * ($gstRatePct/100));
    return [$amount, $tax];
  }
}
```

### Invoice model (memory-light, PDO)

```php
<?php
namespace App\Models;

use App\Core\DB;
use App\Helpers\GST;

class Invoice {
  public static function paginate(int $page=1, int $per=50): array {
    $off = ($page-1)*$per;
    $pdo = DB::pdo();
    $rows = $pdo->prepare("SELECT * FROM invoices ORDER BY issue_date DESC, id DESC LIMIT :l OFFSET :o");
    $rows->bindValue(':l', $per, \PDO::PARAM_INT);
    $rows->bindValue(':o', $off, \PDO::PARAM_INT);
    $rows->execute();
    $count = $pdo->query("SELECT COUNT(*) c FROM invoices")->fetchColumn();
    return ['data'=>$rows->fetchAll(), 'total'=>(int)$count, 'page'=>$page, 'per'=>$per];
  }

  public static function find(int $id): ?array {
    $pdo = DB::pdo();
    $stmt = $pdo->prepare("SELECT * FROM invoices WHERE id=?");
    $stmt->execute([$id]);
    $inv = $stmt->fetch();
    if (!$inv) return null;
    $lines = $pdo->prepare("SELECT * FROM invoice_lines WHERE invoice_id=? ORDER BY id");
    $lines->execute([$id]);
    $inv['lines'] = $lines->fetchAll();
    return $inv;
  }

  public static function create(array $data, array $lines): int {
    return DB::tx(function($pdo) use ($data, $lines) {
      $stmt = $pdo->prepare("INSERT INTO invoices (number, issue_date, party_id, place_of_supply, taxable_value, tax_value, total_value) VALUES (?,?,?,?,?,?,?)");
      $stmt->execute([$data['number'],$data['issue_date'],$data['party_id'],$data['place_of_supply'],0,0,0]);
      $id = (int)$pdo->lastInsertId();

      $taxable = 0.0; $tax = 0.0;
      $lineStmt = $pdo->prepare("INSERT INTO invoice_lines (invoice_id, item_id, description, qty, rate, gst_rate, amount, tax) VALUES (?,?,?,?,?,?,?,?)");
      foreach ($lines as $ln) {
        [$amount, $taxLine] = \App\Helpers\GST::computeLine((float)$ln['qty'], (float)$ln['rate'], (float)$ln['gst_rate']);
        $taxable += $amount; $tax += $taxLine;
        $lineStmt->execute([$id,$ln['item_id'],$ln['description'],$ln['qty'],$ln['rate'],$ln['gst_rate'],$amount,$taxLine]);
      }

      $intra = self::isIntraState($data['place_of_supply'] ?? null);
      $split = \App\Helpers\GST::split($tax, $intra);
      $total = \App\Helpers\Money::round2($taxable + $tax);

      $upd = $pdo->prepare("UPDATE invoices SET taxable_value=?, tax_value=?, total_value=?, cgst=?, sgst=?, igst=? WHERE id=?");
      $upd->execute([$taxable, $tax, $total, $split['cgst'], $split['sgst'], $split['igst'], $id]);

      return $id;
    });
  }

  private static function isIntraState(?string $pos): bool {
    // Simplify: treat 'UP' or '09' as intra for Mawana. Adapt to your schema.
    if (!$pos) return true;
    $pos = strtoupper($pos);
    return ($pos === 'UP' || $pos === '09' || str_contains($pos, 'UTTAR PRADESH'));
  }
}
```

### Actions/invoices/ImportPreview.php

```php
<?php
namespace App\Actions\invoices;

use App\Helpers\Validator;
use App\Helpers\Money;

class ImportPreview {
  public function run(array $rows): array {
    $errors = []; $ok = []; $sum = ['count'=>0,'taxable'=>0.0,'tax'=>0.0];
    foreach ($rows as $i => $r) {
      $e = [];
      if (!Validator::date($r['issue_date'])) $e[] = "invalid date";
      if (!Validator::num($r['qty'])) $e[] = "qty";
      if (!Validator::num($r['rate'])) $e[] = "rate";
      if (!Validator::num($r['gst_rate'])) $e[] = "gst_rate";
      if ($e) {
        $errors[] = ['row'=>$i+1,'errors'=>$e,'sample'=>$r];
        continue;
      }
      $amount = Money::round2($r['qty']*$r['rate']);
      $tax = Money::round2($amount * ($r['gst_rate']/100));
      $sum['count']++; $sum['taxable'] += $amount; $sum['tax'] += $tax;
      $ok[] = $r + ['amount'=>$amount,'tax'=>$tax];
    }
    $token = bin2hex(random_bytes(16));
    $_SESSION['import_token'] = $token;
    $_SESSION['import_rows']  = $ok;
    return ['token'=>$token, 'valid'=>$ok, 'errors'=>$errors, 'summary'=>[
      'count'=>$sum['count'],
      'taxable'=>Money::round2($sum['taxable']),
      'tax'=>Money::round2($sum['tax'])
    ]];
  }
}
```

### Actions/invoices/ImportCommit.php

```php
<?php
namespace App\Actions\invoices;

use App\Core\DB;
use App\Models\Invoice;

class ImportCommit {
  public function run(string $token, string $filePath): array {
    if (!isset($_SESSION['import_token']) || $token !== $_SESSION['import_token']) {
      return ['ok'=>false, 'message'=>'Preview expired or token mismatch'];
    }
    $rows = $_SESSION['import_rows'] ?? [];
    if (!$rows) return ['ok'=>false, 'message'=>'No rows to import'];

    $created = 0;
    DB::tx(function() use ($rows, &$created) {
      // Here: group rows by invoice number+date+party to form invoices with multiple lines
      $groups = [];
      foreach ($rows as $r) {
        $key = $r['number'].'|'.$r['issue_date'].'|'.$r['party'];
        $groups[$key][] = $r;
      }
      foreach ($groups as $key => $lines) {
        $first = $lines[0];
        $data = [
          'number' => $first['number'],
          'issue_date' => $first['issue_date'],
          'party_id' => $this->resolveParty($first['party']),
          'place_of_supply' => $first['place_of_supply'] ?? 'UP',
        ];
        $linePayload = array_map(function($l) {
          return [
            'item_id' => $this->resolveItem($l['item']),
            'description' => $l['item'],
            'qty' => (float)$l['qty'],
            'rate' => (float)$l['rate'],
            'gst_rate' => (float)$l['gst_rate'],
          ];
        }, $lines);
        Invoice::create($data, $linePayload);
        $created++;
      }
    });

    unset($_SESSION['import_token'], $_SESSION['import_rows']);
    return ['ok'=>true, 'created'=>$created];
  }

  private function resolveParty(string $name): int {
    // Upsert minimal party by name
    $pdo = \App\Core\DB::pdo();
    $sel = $pdo->prepare("SELECT id FROM parties WHERE name=?");
    $sel->execute([$name]);
    $id = $sel->fetchColumn();
    if ($id) return (int)$id;
    $ins = $pdo->prepare("INSERT INTO parties (name, state_code) VALUES (?, ?)");
    $ins->execute([$name, '09']);
    return (int)$pdo->lastInsertId();
  }

  private function resolveItem(string $name): int {
    $pdo = \App\Core\DB::pdo();
    $sel = $pdo->prepare("SELECT id FROM items WHERE name=?");
    $sel->execute([$name]);
    $id = $sel->fetchColumn();
    if ($id) return (int)$id;
    $ins = $pdo->prepare("INSERT INTO items (name, hsn) VALUES (?, ?)");
    $ins->execute([$name, null]);
    return (int)$pdo->lastInsertId();
  }
}
```

```php
namespace App\Controllers;

    $q = $pdo->query("
      SELECT
        DATE_FORMAT(issue_date,'%Y-%m') as period,
        SUM(taxable_value) as taxable,
        SUM(cgst) as cgst,
        SUM(sgst) as sgst,
        SUM(igst) as igst,
        SUM(total_value) as total

      ORDER BY period DESC
      LIMIT 24
    ");

      FROM invoices
      GROUP BY period
      ORDER BY period DESC
      LIMIT 24
```

## Minimal views and UX notes

- **Layout:** Dark, eye-friendly, single CSS file, no frontend frameworks.
- **List screens:** Pagination, fast search by number/date/party.
- **Forms:** Safe defaults, client-side number inputs, server validation.
- **Import pages:** Two-step flow (Preview -> Commit) with error table and a disabled Commit button until token is present.
- **Refresh-safe:** ImportCommit reads from server-side session token and re-validates; transactions ensure atomicity; idempotency by unique constraint on invoices(number, party_id, issue_date) if you choose.

Example unique index (optional):

```sql
ALTER TABLE invoices ADD UNIQUE KEY uniq_inv (number, party_id, issue_date);
```

If a duplicate occurs during commit, catch and skip or update as per your policy.
