#!/bin/bash

DB="gst_accounting"
USER="gstwork"
PASS="gstwork@123"
HOST="127.0.0.1"
PORT="3306"

# Output directories
CLEAN_DIR="/var/www/html/bharat_accounting/database/clean_sql_data"
MIGRABLE_DIR="/var/www/html/bharat_accounting/database/migrable_sql_data"

# Ensure directories exist
mkdir -p "$CLEAN_DIR" "$MIGRABLE_DIR"

# Combined files
CLEAN_ALL="$CLEAN_DIR/all_tables.sql"
MIGRABLE_ALL="$MIGRABLE_DIR/all_tables.sql"

# Reset combined files
> "$CLEAN_ALL"
> "$MIGRABLE_ALL"

# Get all tables
tables=$(mysql -u $USER -p$PASS -h $HOST -P $PORT -N -e "SHOW TABLES FROM $DB;")

for t in $tables; do
  echo "Exporting $t..."

  # Clean schema (minimal, no wrappers, no triggers)
  mysqldump -u $USER -p$PASS -h $HOST -P $PORT \
      --skip-comments \
    --skip-set-charset \
    --compact \
    $DB $t > "$CLEAN_DIR/$t.sql"

  # Append to combined clean schema
  cat "$CLEAN_DIR/$t.sql" >> "$CLEAN_ALL"
  echo -e "\n" >> "$CLEAN_ALL"

  # Migrable schema (full dump with wrappers, triggers, etc.)
  mysqldump -u $USER -p$PASS -h $HOST -P $PORT \
       --routines \
    --triggers \
    $DB $t > "$MIGRABLE_DIR/$t.sql"

  # Append to combined migrable schema
  cat "$MIGRABLE_DIR/$t.sql" >> "$MIGRABLE_ALL"
  echo -e "\n" >> "$MIGRABLE_ALL"
done

echo "✅ Export complete."
echo "Clean data: $CLEAN_DIR (per-table + all_tables.sql)"
echo "Migrable data: $MIGRABLE_DIR (per-table + all_tables.sql)"
